<h2>Help Desk Chatbot</h2>

<div id="chat">Hello! How can I help you?</div>


<input id="msg" placeholder="Enter your question">

<button onclick="reply()">Send</button>


<script>

function reply() {

    let m = document.getElementById("msg").value.toLowerCase();

    let r = "Sorry, I don't understand.";


    if (m.includes("hi"))

        r = "Hello! How can I help you?";

    else if (m.includes("password"))

        r = "Click Forgot Password to reset it.";

    else if (m.includes("internet"))

        r = "Please check your internet connection.";

    else if (m.includes("bye"))

        r = "Goodbye!";


    document.getElementById("chat").innerHTML = r;

    document.getElementById("msg").value = "";

}

</script>
